    @RestController
    @RequestMapping("/api/ingressos")
    public class IngressoController {

        private final IngressoRepository repository;

        public IngressoController(IngressoRepository repository) {
            this.repository = repository;
        }

        @PostMapping
        public Ingresso comprar(@RequestBody Ingresso ingresso) {
            if (ingresso.getIdade() < 12) {
                ingresso.setPreco(ingresso.getPreco() * 0.5);
            }
            return repository.save(ingresso);
        }
    }

E) A regra da meia-entrada está no Controller, que ainda acessa o repository
   direto; essa lógica deveria ficar em um Service
